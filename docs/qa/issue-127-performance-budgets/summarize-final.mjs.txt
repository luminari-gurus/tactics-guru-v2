import fs from 'node:fs';
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import {evaluateCapture,LIMITS,FLOAT_NOISE_MS} from '../scripts/measure-fit-evaluate.ts';
const qa='docs/qa/issue-127-performance-budgets/';
const expected={
  'historical-release':'13418626fab4cc40609cd2255d54800c15e34805',
  'baseline-edge':'f1184d5ab474a97df418183edb4f56f4831a4366',
  'candidate-edge':'f28f33181f2d1d4cb6eacf284e7753e56c674397',
  'candidate-preview':'f28f33181f2d1d4cb6eacf284e7753e56c674397',
};
const summaries={},browsers=new Set(),hashes={};
const source=JSON.parse(fs.readFileSync(qa+'current-source-binding.json'));
for(const [prefix,commit] of Object.entries(expected)) for(const route of ['battle','run']){
  const name=prefix+'-'+route,raw=fs.readFileSync(qa+name+'.json'),capture=JSON.parse(raw);
  assert.equal(capture.measuredBuild.commit,commit);assert.equal(capture.measuredBuild.dirty,false);
  assert.deepEqual(capture.collectorSha256,source.collectorSha256);
  const result=evaluateCapture(capture),saved=JSON.parse(fs.readFileSync(qa+name+'-gates.json'));
  assert.deepEqual(result.rows,saved.rows);assert.deepEqual(result.captureFailures,[]);
  assert.equal(result.rows.length,18);browsers.add(capture.browserVersion);
  const cold=result.rows.filter(r=>r.cache==='cold'),warm=result.rows.filter(r=>r.cache==='warm');
  const violations={};for(const row of result.rows)for(const v of row.violations)violations[v.metric]=(violations[v.metric]??0)+1;
  summaries[name]={commit,samples:18,captureFailures:0,budgetFailedSamples:result.failedSamples,
    coldCodeBytes:{min:Math.min(...cold.map(r=>r.values.codeBytes)),max:Math.max(...cold.map(r=>r.values.codeBytes))},
    worstColdUsableMs:Math.max(...cold.map(r=>r.values.usableMs)),worstWarmUsableMs:Math.max(...warm.map(r=>r.values.usableMs)),
    worstAssetBytes:Math.max(...cold.map(r=>r.values.assetBytes)),worstIdleP50Ms:Math.max(...result.rows.map(r=>r.values.idleP50Ms)),
    worstIdleP95Ms:Math.max(...result.rows.map(r=>r.values.idleP95Ms)),
    worstInteractionP95Ms:Math.max(...capture.samples.flatMap(s=>s.diagnostics.workloads.filter(w=>['pan','zoom','move'].includes(w.action)).map(w=>w.frames.p95Ms))),
    violations};
  hashes[name+'.json']=createHash('sha256').update(raw).digest('hex');
}
assert.deepEqual([...browsers],['155.0.4283.45']);
for(const [name,hash] of Object.entries(source.collectorSha256)){
  assert.equal(createHash('sha256').update(fs.readFileSync(qa+source.collectorSourceDirectory+'/'+name+'.txt')).digest('hex'),hash);
}
const result={browser:[...browsers][0],samples:144,limits:LIMITS,floatNoiseToleranceMs:FLOAT_NOISE_MS,
  localCandidateAllGatesPass:['battle','run'].every(route=>summaries['candidate-edge-'+route].budgetFailedSamples===0),
  deployedColdCodePass:['battle','run'].every(route=>summaries['candidate-preview-'+route].coldCodeBytes.max<=LIMITS.L3_cold_code_bytes),
  deployedAllGatesPass:['battle','run'].every(route=>summaries['candidate-preview-'+route].budgetFailedSamples===0),
  localColdCodeSaving:{battle:summaries['baseline-edge-battle'].coldCodeBytes.max-summaries['candidate-edge-battle'].coldCodeBytes.max,
    run:summaries['baseline-edge-run'].coldCodeBytes.max-summaries['candidate-edge-run'].coldCodeBytes.max},
  captures:summaries,rawCaptureSha256:hashes};
fs.writeFileSync(qa+'final-performance-summary.json',JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify(result,null,2));
