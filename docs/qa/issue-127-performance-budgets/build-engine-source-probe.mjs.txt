import {build} from 'vite';
import {resolve,dirname} from 'node:path';
import {createRequire} from 'node:module';
const root=dirname(createRequire(import.meta.url).resolve('phaser/package.json'));
const flags={CANVAS_RENDERER:true,WEBGL_RENDERER:true,WEBGL_DEBUG:false,FEATURE_SOUND:true};
await build({resolve:{alias:[{find:/^phaser$/,replacement:resolve(root,'src/phaser-arcade-physics.js')}]},
 define:{global:'globalThis'},build:{outDir:'tmp/engine-source-probe-dist'},plugins:[{
 name:'stock-phaser-source-flags',enforce:'pre',
 transform(code,id){if(!id.replaceAll('\\','/').includes('/phaser/src/'))return;
  return code.replace(/\btypeof (CANVAS_RENDERER|WEBGL_RENDERER|WEBGL_DEBUG|FEATURE_SOUND)\b/g,(_,flag)=>String(flags[flag]));}
}]});
