foreach ($captureSpec in $captureSpecs) {
  $receiptPath = "docs/qa/issue-127-performance-budgets/$($captureSpec.Name).json"
  $evaluationPath = "docs/qa/issue-127-performance-budgets/$($captureSpec.Name)-gates.json"
  if ((Test-Path -LiteralPath $receiptPath) -or (Test-Path -LiteralPath $evaluationPath)) { throw "Preserve the existing attempt: $receiptPath" }
  node --experimental-strip-types scripts/measure-fit.ts $receiptPath --scene battle --url $captureSpec.Url *> "tmp/$($captureSpec.Name).log"
  $captureExit = $LASTEXITCODE
  if ($captureExit -ne 0) { throw "Invalid capture retained at $receiptPath; inspect its log before continuing." }
  node -e "const r=JSON.parse(require('fs').readFileSync(process.argv[1]));if(r.measuredBuild?.commit!==process.argv[2]||r.measuredBuild.dirty||r.samples.length!==18||r.failures.length)throw Error('Unexpected source or incomplete capture');" $receiptPath $captureSpec.Commit
  if ($LASTEXITCODE -ne 0) { throw "Source/completeness mismatch retained at $receiptPath" }
  node --experimental-strip-types scripts/evaluate-fit.ts $receiptPath $evaluationPath
  # Exit 1 from the evaluator is an actual budget result, not grounds to replace a sample.
