import fs from 'node:fs';
import path from 'node:path';
import {createHash} from 'node:crypto';
import {execFileSync} from 'node:child_process';
import assert from 'node:assert/strict';
const qa='docs/qa/issue-127-performance-budgets/',candidate='06ad2435c28c96b1614eb083acd8f953f76c8fe7',baseline='f28f33181f2d1d4cb6eacf284e7753e56c674397';
const git=(...args)=>execFileSync('git',args,{encoding:'utf8'}).trim();
const blob=(ref,file)=>git('rev-parse',ref+':'+file);
const hash=data=>createHash('sha256').update(data).digest('hex');
const sharedInputs={};for(const file of ['public','index.html','package.json','package-lock.json']){
  assert.equal(blob(candidate,file),blob(baseline,file));sharedInputs[file]=blob(candidate,file);
}
const sourceDifferences=git('diff','--name-only',baseline,candidate,'--','src').split('\n');
assert.deepEqual(sourceDifferences,['src/phaser/contentLoading.ts']);
const files={};for(const file of ['vite.config.ts','src/phaser/contentLoading.ts','tests/public-asset-base.spec.ts']){
  assert.equal(git('diff',candidate,'--',file),'');
  files[file]={gitBlob:blob(candidate,file),checkoutSha256:hash(fs.readFileSync(file))};
}
function inventory(root,relative=''){
  return fs.readdirSync(path.join(root,relative),{withFileTypes:true}).flatMap(entry=>{
    const name=path.join(relative,entry.name),file=path.join(root,name);
    if(entry.isDirectory())return inventory(root,name);
    const bytes=fs.readFileSync(file);return [{path:name.replaceAll('\\','/'),bytes:bytes.length,sha256:hash(bytes)}];
  }).sort((a,b)=>a.path.localeCompare(b.path));
}
for(const kind of ['final','prefix']){
  const root='tmp/startup-'+kind+'-dist';
  fs.writeFileSync(qa+'startup-'+kind+'-build-files.json',JSON.stringify({candidate,base:kind==='prefix'?'/tactics/':'/',files:inventory(root)},null,2)+'\n');
}
const prior=JSON.parse(fs.readFileSync(qa+'current-source-binding.json'));
for(const [file,expected]of Object.entries(prior.collectorSha256))assert.equal(hash(fs.readFileSync('scripts/'+file)),expected);
fs.writeFileSync(qa+'startup-source-binding.json',JSON.stringify({baseline,candidate,main:git('rev-parse','origin/main'),
  sharedInputs,sourceDifferences,files,collectorSha256:prior.collectorSha256,collectorSourceDirectory:prior.collectorSourceDirectory,
  runtimeDifference:'Route-specific build HTML hints and the existing content loader selecting Phaser HTMLImageElement transport. No other application/public/package changes relative to f28f331.',
  preview:'https://b0761a21.tactics-guru-v2.pages.dev/',local:'http://127.0.0.1:4199/',
  buildNote:'Root and prefix artifacts were built from clean 06ad243 before additional QA files. Collector sourceDirty reflects accumulated QA, not dirty served artifacts.'},null,2)+'\n');
fs.copyFileSync('tmp/startup-deployment.json',qa+'startup-deployment.json');
console.log('Source, shared inputs, collector hashes and both frozen build inventories verified.');
