import fs from 'node:fs';
import path from 'node:path';
import crypto from 'node:crypto';
import assert from 'node:assert/strict';
const origin='https://f2ba4504.tactics-guru-v2.pages.dev';
const hash=b=>crypto.createHash('sha256').update(b).digest('hex');
const paths=fs.readFileSync('dist/_headers','utf8').trim().split('\n\n').map(block=>block.split('\n')[0]);
const code=await Promise.all(paths.map(async file=>{const r=await fetch(origin+file);const bytes=Buffer.from(await r.arrayBuffer()),local=fs.readFileSync('dist'+file);assert.equal(r.status,200);assert.equal(r.headers.get('cache-control'),'public, max-age=31536000, immutable');assert.equal(hash(bytes),hash(local));return {file,status:r.status,cacheControl:r.headers.get('cache-control'),bytes:bytes.length,sha256:hash(bytes)};}));
const fixed=await Promise.all(['/','/battle/terrain/grass.png','/audio/battle/move.mp3'].map(async file=>{const r=await fetch(origin+file,{method:'HEAD'});assert.equal(r.status,200);assert.equal(r.headers.get('cache-control'),'public, max-age=0, must-revalidate');return {file,status:r.status,cacheControl:r.headers.get('cache-control')};}));
function inventory(dir){return fs.readdirSync(dir,{withFileTypes:true}).flatMap(entry=>{const file=path.join(dir,entry.name);if(entry.isDirectory())return inventory(file);const bytes=fs.readFileSync(file);return [{path:file.replaceAll('\\','/'),bytes:bytes.length,sha256:hash(bytes)}];});}
fs.writeFileSync('tmp/review/pr137-preview-verification.json',JSON.stringify({source:'6e615dca32f19bdde9afab5f3233ba089aaa74b4',origin,verifiedAt:new Date().toISOString(),code,fixed},null,2)+'\n');
fs.writeFileSync('tmp/review/pr137-build-inventory.json',JSON.stringify(inventory('dist'),null,2)+'\n');
console.log(`Verified ${code.length} exact code files and three HTML/fixed-asset policies.`);
