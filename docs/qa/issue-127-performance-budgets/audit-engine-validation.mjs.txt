import fs from 'node:fs';
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';

const reports={};
for(const [suite,count]of Object.entries(expected)){
 const path=`tmp/engine-${suite}.json`,raw=fs.readFileSync(path),report=JSON.parse(raw);
 const tests=[];const visit=s=>{for(const spec of s.specs??[])for(const test of spec.tests)tests.push({title:spec.title,...test});for(const child of s.suites??[])visit(child)};visit(report);
 assert.equal(tests.length,count);assert.equal(report.stats.expected,count);assert.equal(report.stats.unexpected,0);assert.equal(report.stats.flaky,0);assert.equal(report.stats.skipped,0);
 assert(tests.every(t=>t.results.length===1&&t.results[0].status==='passed'&&t.results[0].retry===0));
 reports[suite]={file:path,stats:report.stats,tests:count,skips:0,retries:0,sha256:createHash('sha256').update(raw).digest('hex')};
}
const binding=JSON.parse(fs.readFileSync('tmp/engine-repaired-source-binding.json'));
const failed=JSON.parse(fs.readFileSync('tmp/engine-encounters.json'));assert.equal(failed.stats.unexpected,18);

console.log('707 current functional cases passed, no skips/retries; initial 18 failures and zero-test invocation retained.');
